function(_adc700_json_array output json)
  string(JSON _type ERROR_VARIABLE _type_error TYPE "${json}" ${ARGN})
  if(_type_error OR NOT _type STREQUAL "ARRAY")
    message(FATAL_ERROR "ADC-700 toolchain receipt field is not an array: ${ARGN}")
  endif()
  string(JSON _length ERROR_VARIABLE _json_error LENGTH "${json}" ${ARGN})
  if(_json_error OR NOT _length MATCHES "^[0-9]+$")
    message(FATAL_ERROR "ADC-700 toolchain receipt array is not valid: ${ARGN}")
  endif()
  set(_values "")
  if(_length GREATER 0)
    math(EXPR _last "${_length} - 1")
    foreach(_index RANGE 0 ${_last})
      string(JSON _item_type ERROR_VARIABLE _item_type_error TYPE "${json}" ${ARGN} ${_index})
      string(JSON _value ERROR_VARIABLE _item_error GET "${json}" ${ARGN} ${_index})
      if(_item_type_error OR _item_error OR NOT _item_type STREQUAL "STRING" OR NOT _value)
        message(FATAL_ERROR "ADC-700 toolchain receipt array item is invalid: ${ARGN}[${_index}]")
      endif()
      list(APPEND _values "${_value}")
    endforeach()
  endif()
  set(${output} "${_values}" PARENT_SCOPE)
endfunction()

function(_adc700_json_object_keys output json)
  string(JSON _length ERROR_VARIABLE _length_error LENGTH "${json}" ${ARGN})
  if(_length_error OR NOT _length MATCHES "^[0-9]+$")
    message(FATAL_ERROR "ADC-700 toolchain receipt object is not valid: ${ARGN}")
  endif()
  set(_keys "")
  if(_length GREATER 0)
    math(EXPR _last "${_length} - 1")
    foreach(_index RANGE 0 ${_last})
      string(JSON _key ERROR_VARIABLE _key_error MEMBER "${json}" ${ARGN} ${_index})
      if(_key_error OR NOT _key)
        message(FATAL_ERROR "ADC-700 toolchain receipt object member is invalid: ${ARGN}")
      endif()
      list(APPEND _keys "${_key}")
    endforeach()
  endif()
  set(${output} "${_keys}" PARENT_SCOPE)
endfunction()

function(_adc700_require_object json path)
  string(JSON _type ERROR_VARIABLE _type_error TYPE "${json}" ${path})
  if(_type_error OR NOT _type STREQUAL "OBJECT")
    message(FATAL_ERROR "ADC-700 toolchain receipt field is not an object: ${path}")
  endif()
  _adc700_json_object_keys(_actual_keys "${json}" ${path})
  set(_expected_keys ${ARGN})
  list(SORT _actual_keys)
  list(SORT _expected_keys)
  if(NOT "${_actual_keys}" STREQUAL "${_expected_keys}")
    message(FATAL_ERROR
      "ADC-700 toolchain receipt object has missing/extra fields at ${path}: "
      "expected=${_expected_keys}, observed=${_actual_keys}")
  endif()
endfunction()

function(_adc700_json_string output json)
  string(JSON _type ERROR_VARIABLE _type_error TYPE "${json}" ${ARGN})
  if(_type_error OR NOT _type STREQUAL "STRING")
    message(FATAL_ERROR "ADC-700 toolchain receipt field is not text: ${ARGN}")
  endif()
  string(JSON _value ERROR_VARIABLE _value_error GET "${json}" ${ARGN})
  if(_value_error OR NOT _value)
    message(FATAL_ERROR "ADC-700 toolchain receipt field is empty: ${ARGN}")
  endif()
  set(${output} "${_value}" PARENT_SCOPE)
endfunction()

function(_adc700_require_hex value expected_length label)
  # CMake's regex dialect does not implement ``{n}`` quantifiers.  Validate
  # length separately so hashes/revisions cannot be accepted as prefixes.
  string(LENGTH "${value}" _actual_length)
  if(NOT _actual_length EQUAL ${expected_length} OR
     NOT "${value}" MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR
      "ADC-700 ${label} must be exactly ${expected_length} hexadecimal characters")
  endif()
endfunction()

function(_adc700_require_integer json expected_value)
  string(JSON _type ERROR_VARIABLE _type_error TYPE "${json}" ${ARGN})
  string(JSON _value ERROR_VARIABLE _value_error GET "${json}" ${ARGN})
  if(_type_error OR _value_error OR NOT _type STREQUAL "NUMBER" OR
     NOT "${_value}" MATCHES "^[0-9]+$" OR NOT _value EQUAL ${expected_value})
    message(FATAL_ERROR
      "ADC-700 ${ARGN} must be the exact integer ${expected_value}")
  endif()
endfunction()

function(_adc700_json_quote output value)
  string(FIND "${value}" ";" _semicolon)
  if(NOT _semicolon EQUAL -1)
    message(FATAL_ERROR "ADC-700 toolchain receipt value cannot be JSON-serialized")
  endif()
  # Command provenance (notably compiler --version) is legitimately multi-line.
  # Escape JSON control characters after protecting literal backslashes so the
  # probe remains valid JSON without normalizing or truncating the value.
  set(_escaped "${value}")
  string(ASCII 8 _backspace)
  string(ASCII 12 _formfeed)
  string(REPLACE "\\" "\\\\" _escaped "${_escaped}")
  string(REPLACE "\"" "\\\"" _escaped "${_escaped}")
  string(REPLACE "${_backspace}" "\\b" _escaped "${_escaped}")
  string(REPLACE "${_formfeed}" "\\f" _escaped "${_escaped}")
  string(REPLACE "\n" "\\n" _escaped "${_escaped}")
  string(REPLACE "\r" "\\r" _escaped "${_escaped}")
  string(REPLACE "\t" "\\t" _escaped "${_escaped}")
  set(${output} "\"${_escaped}\"" PARENT_SCOPE)
endfunction()

function(_adc700_json_string_array output)
  set(_json "[")
  foreach(_value IN LISTS ARGN)
    _adc700_json_quote(_quoted "${_value}")
    if(NOT _json STREQUAL "[")
      string(APPEND _json ",")
    endif()
    string(APPEND _json "${_quoted}")
  endforeach()
  string(APPEND _json "]")
  set(${output} "${_json}" PARENT_SCOPE)
endfunction()

function(_adc700_json_file_rows output paths hashes)
  list(LENGTH ${paths} _path_count)
  list(LENGTH ${hashes} _hash_count)
  if(NOT _path_count EQUAL _hash_count)
    message(FATAL_ERROR "ADC-700 toolchain receipt path/hash arity differs")
  endif()
  set(_json "[")
  if(_path_count GREATER 0)
    math(EXPR _last "${_path_count} - 1")
    foreach(_index RANGE 0 ${_last})
      list(GET ${paths} ${_index} _path)
      list(GET ${hashes} ${_index} _hash)
      _adc700_json_quote(_quoted_path "${_path}")
      _adc700_json_quote(_quoted_hash "${_hash}")
      if(NOT _json STREQUAL "[")
        string(APPEND _json ",")
      endif()
      string(APPEND _json "{\"path\":${_quoted_path},\"sha256\":${_quoted_hash}}")
    endforeach()
  endif()
  string(APPEND _json "]")
  set(${output} "${_json}" PARENT_SCOPE)
endfunction()

function(_adc700_target_property output target property)
  if(NOT TARGET ${target})
    message(FATAL_ERROR "ADC-700 expected target does not exist: ${target}")
  endif()
  get_target_property(_values ${target} ${property})
  if(_values MATCHES "-NOTFOUND$")
    set(_values "")
  endif()
  set(_clean "")
  foreach(_value IN LISTS _values)
    string(FIND "${_value}" "\n" _line_feed)
    string(FIND "${_value}" "\r" _carriage_return)
    if(_value MATCHES "\\$<" OR _line_feed GREATER_EQUAL 0 OR _carriage_return GREATER_EQUAL 0)
      message(FATAL_ERROR
        "ADC-700 target ${target} exposes a non-replayable ${property}: ${_value}")
    endif()
    list(APPEND _clean "${_value}")
  endforeach()
  set(${output} "${_clean}" PARENT_SCOPE)
endfunction()

function(_adc700_normalize_directories output)
  set(_clean "")
  foreach(_value IN LISTS ARGN)
    cmake_path(NORMAL_PATH _value OUTPUT_VARIABLE _normalized)
    if(NOT IS_ABSOLUTE "${_normalized}" OR NOT IS_DIRECTORY "${_normalized}")
      message(FATAL_ERROR "ADC-700 target include is not an absolute directory: ${_value}")
    endif()
    list(APPEND _clean "${_normalized}")
  endforeach()
  set(${output} "${_clean}" PARENT_SCOPE)
endfunction()

function(_adc700_normalize_kokkos_directories output)
  set(_clean "")
  foreach(_value IN LISTS ARGN)
    if(_value MATCHES "^\\$<BUILD_INTERFACE:(.*)>$")
      set(_value "${CMAKE_MATCH_1}")
    elseif(_value MATCHES "^\\$<INSTALL_INTERFACE:.*>$")
      continue()
    elseif(_value MATCHES "\\$<")
      message(FATAL_ERROR "ADC-700 Kokkos include is not replayable: ${_value}")
    endif()
    cmake_path(NORMAL_PATH _value OUTPUT_VARIABLE _normalized)
    if(NOT IS_ABSOLUTE "${_normalized}" OR NOT IS_DIRECTORY "${_normalized}")
      message(FATAL_ERROR "ADC-700 Kokkos include is not an absolute directory: ${_value}")
    endif()
    list(APPEND _clean "${_normalized}")
  endforeach()
  list(REMOVE_DUPLICATES _clean)
  set(${output} "${_clean}" PARENT_SCOPE)
endfunction()

function(_adc700_verify_file output_hash path expected_hash label)
  if(NOT IS_ABSOLUTE "${path}" OR IS_SYMLINK "${path}" OR
     NOT EXISTS "${path}" OR IS_DIRECTORY "${path}")
    message(FATAL_ERROR "ADC-700 ${label} is not an absolute regular file: ${path}")
  endif()
  file(SHA256 "${path}" _observed)
  if(NOT _observed STREQUAL "${expected_hash}")
    message(FATAL_ERROR "ADC-700 ${label} SHA-256 changed: ${path}")
  endif()
  set(${output_hash} "${_observed}" PARENT_SCOPE)
endfunction()

function(_adc700_parse_file_rows output_paths output_hashes json)
  # ``section`` is a JSON path (for example ``mpi headers``), not one scalar argument.
  # Preserve every path component so nested receipt arrays are actually parsed.
  string(JSON _type ERROR_VARIABLE _type_error TYPE "${json}" ${ARGN})
  string(JSON _length ERROR_VARIABLE _length_error LENGTH "${json}" ${ARGN})
  if(_type_error OR _length_error OR NOT _type STREQUAL "ARRAY" OR
     NOT _length MATCHES "^[0-9]+$")
    message(FATAL_ERROR "ADC-700 receipt file rows are not an array: ${ARGN}")
  endif()
  set(_paths "")
  set(_hashes "")
  if(_length GREATER 0)
    math(EXPR _last "${_length} - 1")
    foreach(_index RANGE 0 ${_last})
      string(JSON _row_type ERROR_VARIABLE _row_type_error TYPE "${json}" ${ARGN} ${_index})
      if(_row_type_error OR NOT _row_type STREQUAL "OBJECT")
        message(FATAL_ERROR "ADC-700 receipt file row is not an object: ${ARGN}[${_index}]")
      endif()
      _adc700_json_object_keys(_row_keys "${json}" ${ARGN} ${_index})
      set(_expected path sha256)
      list(SORT _row_keys)
      list(SORT _expected)
      if(NOT "${_row_keys}" STREQUAL "${_expected}")
        message(FATAL_ERROR "ADC-700 receipt file row has missing/extra fields: ${ARGN}[${_index}]")
      endif()
      _adc700_json_string(_path "${json}" ${ARGN} ${_index} path)
      _adc700_json_string(_hash "${json}" ${ARGN} ${_index} sha256)
      _adc700_require_hex("${_hash}" 64 "${ARGN}[${_index}] SHA-256")
      _adc700_verify_file(_observed "${_path}" "${_hash}" "${ARGN}[${_index}]")
      list(APPEND _paths "${_path}")
      list(APPEND _hashes "${_observed}")
    endforeach()
  endif()
  set(${output_paths} "${_paths}" PARENT_SCOPE)
  set(${output_hashes} "${_hashes}" PARENT_SCOPE)
endfunction()

function(_adc700_kokkos_abi output includes header_paths header_hashes)
  set(_material "")
  foreach(_include IN LISTS ${includes})
    string(APPEND _material "include=${_include}\n")
    foreach(_name IN ITEMS Kokkos_Core.hpp KokkosCore_config.h)
      set(_header "${_include}/${_name}")
      list(FIND ${header_paths} "${_header}" _index)
      if(_index GREATER_EQUAL 0)
        list(GET ${header_hashes} ${_index} _hash)
        string(APPEND _material "header=${_header};sha256=${_hash}\n")
      endif()
    endforeach()
  endforeach()
  string(SHA256 _digest "${_material}")
  set(${output} "${_digest}" PARENT_SCOPE)
endfunction()

function(_adc700_mpi_abi output compiler standard compile_options compile_definitions link_options
         includes header_paths header_hashes library_paths library_hashes)
  set(_material "compiler=${compiler}\nstandard=${standard}\n")
  foreach(_value IN LISTS ${compile_options})
    string(APPEND _material "compile_option=${_value}\n")
  endforeach()
  foreach(_value IN LISTS ${compile_definitions})
    string(APPEND _material "compile_definition=${_value}\n")
  endforeach()
  foreach(_value IN LISTS ${link_options})
    string(APPEND _material "link_option=${_value}\n")
  endforeach()
  foreach(_include IN LISTS ${includes})
    string(APPEND _material "include=${_include}\n")
    set(_header "${_include}/mpi.h")
    list(FIND ${header_paths} "${_header}" _index)
    if(_index GREATER_EQUAL 0)
      list(GET ${header_hashes} ${_index} _hash)
      string(APPEND _material "header=${_header};sha256=${_hash}\n")
    endif()
  endforeach()
  list(LENGTH ${library_paths} _library_count)
  if(_library_count GREATER 0)
    math(EXPR _last "${_library_count} - 1")
    foreach(_index RANGE 0 ${_last})
      list(GET ${library_paths} ${_index} _path)
      list(GET ${library_hashes} ${_index} _hash)
      string(APPEND _material "library=${_path};sha256=${_hash}\n")
    endforeach()
  endif()
  string(SHA256 _digest "${_material}")
  set(${output} "${_digest}" PARENT_SCOPE)
endfunction()

function(_adc700_command_provenance output_sha output_version executable label)
  if(NOT IS_ABSOLUTE "${executable}" OR IS_SYMLINK "${executable}" OR
     NOT EXISTS "${executable}" OR IS_DIRECTORY "${executable}")
    message(FATAL_ERROR "ADC-700 ${label} is not an absolute regular file: ${executable}")
  endif()
  file(SHA256 "${executable}" _sha256)
  execute_process(
    COMMAND "${executable}" --version
    RESULT_VARIABLE _result
    OUTPUT_VARIABLE _stdout
    ERROR_VARIABLE _stderr
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_STRIP_TRAILING_WHITESPACE)
  if(NOT _result EQUAL 0)
    message(FATAL_ERROR "ADC-700 ${label} --version failed")
  endif()
  string(CONCAT _version "${_stdout}" "\n" "${_stderr}")
  string(STRIP "${_version}" _version)
  if(NOT _version)
    message(FATAL_ERROR "ADC-700 ${label} --version returned no authenticated text")
  endif()
  set(${output_sha} "${_sha256}" PARENT_SCOPE)
  set(${output_version} "${_version}" PARENT_SCOPE)
endfunction()

function(_adc700_probe_target_contract)
  if(NOT POPS_ADC700_TOOLCHAIN_PROBE_OUTPUT OR
     NOT IS_ABSOLUTE "${POPS_ADC700_TOOLCHAIN_PROBE_OUTPUT}")
    message(FATAL_ERROR "ADC-700 toolchain probe requires an absolute output path")
  endif()
  if(NOT TARGET MPI::MPI_CXX OR NOT TARGET Kokkos::kokkos)
    message(FATAL_ERROR "ADC-700 toolchain probe could not materialize MPI::MPI_CXX/Kokkos::kokkos")
  endif()

  # The probe is run with the exact wrapper/root selected for the candidate wheel.  Keep every
  # target property concrete and preserve order: the baseline verifier later compares the same
  # imported targets byte-for-byte instead of accepting a subset of flags.
  file(REAL_PATH "${CMAKE_CXX_COMPILER}" _nvcc_path)
  _adc700_command_provenance(_nvcc_sha _nvcc_version "${_nvcc_path}" "NVCC wrapper")
  _adc700_target_property(_mpi_includes_raw MPI::MPI_CXX INTERFACE_INCLUDE_DIRECTORIES)
  _adc700_target_property(_mpi_compile_options MPI::MPI_CXX INTERFACE_COMPILE_OPTIONS)
  _adc700_target_property(_mpi_compile_definitions MPI::MPI_CXX INTERFACE_COMPILE_DEFINITIONS)
  _adc700_target_property(_mpi_link_options MPI::MPI_CXX INTERFACE_LINK_OPTIONS)
  _adc700_target_property(_mpi_libraries_raw MPI::MPI_CXX INTERFACE_LINK_LIBRARIES)
  _adc700_normalize_directories(_mpi_includes ${_mpi_includes_raw})
  set(_mpi_libraries "")
  set(_mpi_library_hashes "")
  foreach(_library IN LISTS _mpi_libraries_raw)
    cmake_path(NORMAL_PATH _library OUTPUT_VARIABLE _library_normalized)
    if(NOT IS_ABSOLUTE "${_library_normalized}" OR IS_SYMLINK "${_library_normalized}" OR
       NOT EXISTS "${_library_normalized}" OR IS_DIRECTORY "${_library_normalized}")
      message(FATAL_ERROR
        "ADC-700 MPI::MPI_CXX exposes a non-authenticatable library: ${_library}")
    endif()
    file(SHA256 "${_library_normalized}" _library_hash)
    list(APPEND _mpi_libraries "${_library_normalized}")
    list(APPEND _mpi_library_hashes "${_library_hash}")
  endforeach()
  if(NOT _mpi_libraries)
    message(FATAL_ERROR "ADC-700 MPI::MPI_CXX exposes no concrete link libraries")
  endif()
  set(_mpi_headers "")
  set(_mpi_header_hashes "")
  foreach(_include IN LISTS _mpi_includes)
    set(_header "${_include}/mpi.h")
    if(EXISTS "${_header}")
      if(IS_SYMLINK "${_header}" OR IS_DIRECTORY "${_header}")
        message(FATAL_ERROR "ADC-700 MPI mpi.h is not a regular file: ${_header}")
      endif()
      file(SHA256 "${_header}" _header_hash)
      list(APPEND _mpi_headers "${_header}")
      list(APPEND _mpi_header_hashes "${_header_hash}")
    endif()
  endforeach()
  if(NOT _mpi_headers)
    message(FATAL_ERROR "ADC-700 MPI include targets expose no mpi.h")
  endif()
  # Keep the compiler spelling identical to the authenticated path and hash emitted below.
  if(NOT IS_ABSOLUTE "${MPI_CXX_COMPILER}")
    message(FATAL_ERROR "ADC-700 FindMPI compiler path must be absolute")
  endif()
  file(REAL_PATH "${MPI_CXX_COMPILER}" _mpi_compiler_real)
  _adc700_command_provenance(_mpi_compiler_sha _mpi_compiler_version "${_mpi_compiler_real}" "MPI compiler")
  _adc700_mpi_abi(_mpi_abi "${_mpi_compiler_real}" "${MPI_CXX_VERSION}"
    _mpi_compile_options _mpi_compile_definitions _mpi_link_options _mpi_includes
    _mpi_headers _mpi_header_hashes _mpi_libraries _mpi_library_hashes)

  _adc700_target_property(_kokkos_includes_raw Kokkos::kokkos INTERFACE_INCLUDE_DIRECTORIES)
  _adc700_target_property(_kokkos_compile_options Kokkos::kokkos INTERFACE_COMPILE_OPTIONS)
  _adc700_target_property(_kokkos_compile_definitions Kokkos::kokkos INTERFACE_COMPILE_DEFINITIONS)
  _adc700_target_property(_kokkos_link_options Kokkos::kokkos INTERFACE_LINK_OPTIONS)
  _adc700_target_property(_kokkos_libraries Kokkos::kokkos INTERFACE_LINK_LIBRARIES)
  _adc700_normalize_kokkos_directories(_kokkos_includes ${_kokkos_includes_raw})
  set(_kokkos_headers "")
  set(_kokkos_header_hashes "")
  foreach(_include IN LISTS _kokkos_includes)
    foreach(_name IN ITEMS Kokkos_Core.hpp KokkosCore_config.h)
      set(_header "${_include}/${_name}")
      if(EXISTS "${_header}")
        if(IS_SYMLINK "${_header}" OR IS_DIRECTORY "${_header}")
          message(FATAL_ERROR "ADC-700 Kokkos defining header is not a regular file: ${_header}")
        endif()
        file(SHA256 "${_header}" _header_hash)
        list(APPEND _kokkos_headers "${_header}")
        list(APPEND _kokkos_header_hashes "${_header_hash}")
      endif()
    endforeach()
  endforeach()
  list(LENGTH _kokkos_headers _kokkos_header_count)
  if(NOT _kokkos_header_count EQUAL 2)
    message(FATAL_ERROR
      "ADC-700 Kokkos targets must expose exactly Kokkos_Core.hpp and KokkosCore_config.h")
  endif()
  _adc700_kokkos_abi(_kokkos_abi _kokkos_includes _kokkos_headers _kokkos_header_hashes)

  _adc700_json_quote(_nvcc_path_json "${_nvcc_path}")
  _adc700_json_quote(_nvcc_sha_json "${_nvcc_sha}")
  _adc700_json_quote(_nvcc_version_json "${_nvcc_version}")
  set(_nvcc_json
    "{\"path\":${_nvcc_path_json},\"sha256\":${_nvcc_sha_json},\"version\":${_nvcc_version_json}}")
  _adc700_json_string_array(_mpi_compile_options_json ${_mpi_compile_options})
  _adc700_json_string_array(_mpi_compile_definitions_json ${_mpi_compile_definitions})
  _adc700_json_string_array(_mpi_link_options_json ${_mpi_link_options})
  _adc700_json_string_array(_mpi_includes_json ${_mpi_includes})
  _adc700_json_string_array(_mpi_libraries_json ${_mpi_libraries})
  _adc700_json_quote(_mpi_version_json "${MPI_CXX_VERSION}")
  _adc700_json_quote(_mpi_abi_json "${_mpi_abi}")
  _adc700_json_quote(_mpi_compiler_real_json "${_mpi_compiler_real}")
  _adc700_json_quote(_mpi_compiler_version_json "${_mpi_compiler_version}")
  _adc700_json_quote(_mpi_compiler_sha_json "${_mpi_compiler_sha}")
  _adc700_json_file_rows(_mpi_library_rows_json _mpi_libraries _mpi_library_hashes)
  _adc700_json_file_rows(_mpi_header_rows_json _mpi_headers _mpi_header_hashes)
  set(_mpi_json
    "{\"schema_version\":1,\"compiler\":${_mpi_compiler_real_json},\"compiler_version\":${_mpi_compiler_version_json},\"compiler_sha256\":${_mpi_compiler_sha_json},\"abi_sha256\":${_mpi_abi_json},\"standard\":${_mpi_version_json},\"compile_options\":${_mpi_compile_options_json},\"compile_definitions\":${_mpi_compile_definitions_json},\"link_options\":${_mpi_link_options_json},\"link_libraries\":${_mpi_libraries_json},\"include_dirs\":${_mpi_includes_json},\"headers\":${_mpi_header_rows_json},\"libraries\":${_mpi_library_rows_json}}")

  _adc700_json_quote(_kokkos_abi_json "${_kokkos_abi}")
  _adc700_json_string_array(_kokkos_includes_json ${_kokkos_includes})
  _adc700_json_string_array(_kokkos_compile_options_json ${_kokkos_compile_options})
  _adc700_json_string_array(_kokkos_compile_definitions_json ${_kokkos_compile_definitions})
  _adc700_json_string_array(_kokkos_link_options_json ${_kokkos_link_options})
  _adc700_json_string_array(_kokkos_libraries_json ${_kokkos_libraries})
  _adc700_json_file_rows(_kokkos_header_rows_json _kokkos_headers _kokkos_header_hashes)
  set(_kokkos_json
    "{\"schema_version\":1,\"abi_sha256\":${_kokkos_abi_json},\"include_dirs\":${_kokkos_includes_json},\"headers\":${_kokkos_header_rows_json},\"compile_options\":${_kokkos_compile_options_json},\"compile_definitions\":${_kokkos_compile_definitions_json},\"link_options\":${_kokkos_link_options_json},\"link_libraries\":${_kokkos_libraries_json}}")
  set(_contract "{\"nvcc_wrapper\":${_nvcc_json},\"mpi\":${_mpi_json},\"kokkos\":${_kokkos_json}}")
  string(JSON _validated ERROR_VARIABLE _json_error SET "{}" contract "${_contract}")
  if(_json_error)
    message(FATAL_ERROR "ADC-700 toolchain probe generated invalid JSON: ${_json_error}")
  endif()
  file(WRITE "${POPS_ADC700_TOOLCHAIN_PROBE_OUTPUT}" "${_contract}\n")
endfunction()

function(_adc700_real_list output)
  set(_values "")
  foreach(_item IN LISTS ARGN)
    if(NOT _item)
      continue()
    endif()
    file(REAL_PATH "${_item}" _real_path)
    list(APPEND _values "${_real_path}")
  endforeach()
  set(${output} "${_values}" PARENT_SCOPE)
endfunction()

function(_adc700_verify_shared_toolchain)
  if(NOT POPS_ADC700_TOOLCHAIN_RECEIPT OR
     NOT EXISTS "${POPS_ADC700_TOOLCHAIN_RECEIPT}")
    message(FATAL_ERROR
      "ADC-700 baseline requires the candidate toolchain receipt before CMake configuration")
  endif()
  _adc700_require_hex("${POPS_ADC700_TOOLCHAIN_RECEIPT_SHA256}" 64 "toolchain receipt SHA-256")
  _adc700_require_hex("${POPS_ADC700_TOOLCHAIN_REVISION}" 40 "candidate toolchain revision")
  file(REAL_PATH "${POPS_ADC700_TOOLCHAIN_RECEIPT}" _receipt_path)
  file(SHA256 "${_receipt_path}" _receipt_sha256)
  if(NOT _receipt_sha256 STREQUAL "${POPS_ADC700_TOOLCHAIN_RECEIPT_SHA256}")
    message(FATAL_ERROR "ADC-700 toolchain receipt changed before baseline configuration")
  endif()
  file(READ "${_receipt_path}" _receipt_json)
  _adc700_require_object("${_receipt_json}" ""
    nvcc_wrapper mpi kokkos native_loader requested)
  _adc700_require_object("${_receipt_json}" nvcc_wrapper path sha256 version)
  _adc700_require_object("${_receipt_json}" mpi
    schema_version compiler compiler_version compiler_sha256 abi_sha256 standard
    compile_options compile_definitions link_options link_libraries include_dirs headers libraries)
  _adc700_require_object("${_receipt_json}" kokkos
    schema_version abi_sha256 include_dirs headers compile_options compile_definitions
    link_options link_libraries)
  _adc700_require_object("${_receipt_json}" native_loader schema_version compile_definitions)
  _adc700_require_object("${_receipt_json}" requested cxx include std compile_flags link_flags)
  _adc700_require_integer("${_receipt_json}" 1 mpi schema_version)
  _adc700_require_integer("${_receipt_json}" 1 kokkos schema_version)
  _adc700_require_integer("${_receipt_json}" 1 native_loader schema_version)

  # NVCC wrapper identity is checked both against the receipt bytes and against the compiler CMake
  # actually selected.  Version output and SHA are facts, not user-provided labels.
  _adc700_json_string(_nvcc_path "${_receipt_json}" nvcc_wrapper path)
  _adc700_json_string(_nvcc_sha "${_receipt_json}" nvcc_wrapper sha256)
  _adc700_json_string(_nvcc_version "${_receipt_json}" nvcc_wrapper version)
  if(NOT _nvcc_path MATCHES "nvcc_wrapper")
    message(FATAL_ERROR "ADC-700 candidate NVCC wrapper provenance is malformed")
  endif()
  _adc700_require_hex("${_nvcc_sha}" 64 "candidate NVCC wrapper SHA-256")
  _adc700_command_provenance(_observed_nvcc_sha _observed_nvcc_version "${_nvcc_path}" "NVCC wrapper")
  if(NOT _observed_nvcc_sha STREQUAL "${_nvcc_sha}" OR
     NOT _observed_nvcc_version STREQUAL "${_nvcc_version}")
    message(FATAL_ERROR "ADC-700 NVCC wrapper version/SHA differs from receipt")
  endif()
  file(REAL_PATH "${CMAKE_CXX_COMPILER}" _baseline_cxx)
  file(REAL_PATH "${_nvcc_path}" _nvcc_real)
  if(NOT _baseline_cxx STREQUAL "${_nvcc_real}")
    message(FATAL_ERROR
      "ADC-700 baseline compiler differs from receipt NVCC wrapper: ${_baseline_cxx} != ${_nvcc_real}")
  endif()
  if(NOT "${CMAKE_CXX_STANDARD}" STREQUAL "20")
    message(FATAL_ERROR "ADC-700 baseline must compile with exact C++20")
  endif()

  # FindMPI's concrete target is the source of truth for the baseline executable.  Every field of
  # the candidate contract is parsed, re-hashed, and compared with both imported-target properties
  # and FindMPI's variables.  A missing or extra option/library is therefore a hard mismatch.
  if(NOT TARGET MPI::MPI_CXX OR NOT TARGET Kokkos::kokkos)
    message(FATAL_ERROR "ADC-700 baseline could not materialize MPI::MPI_CXX/Kokkos::kokkos")
  endif()
  _adc700_json_string(_mpi_compiler "${_receipt_json}" mpi compiler)
  _adc700_json_string(_mpi_compiler_version "${_receipt_json}" mpi compiler_version)
  _adc700_json_string(_mpi_compiler_sha "${_receipt_json}" mpi compiler_sha256)
  _adc700_json_string(_mpi_abi "${_receipt_json}" mpi abi_sha256)
  _adc700_json_string(_mpi_standard "${_receipt_json}" mpi standard)
  _adc700_require_hex("${_mpi_compiler_sha}" 64 "MPI compiler SHA-256")
  _adc700_require_hex("${_mpi_abi}" 64 "MPI ABI SHA-256")
  _adc700_command_provenance(_observed_mpi_sha _observed_mpi_version "${_mpi_compiler}" "MPI compiler")
  if(NOT _observed_mpi_sha STREQUAL "${_mpi_compiler_sha}" OR
     NOT _observed_mpi_version STREQUAL "${_mpi_compiler_version}")
    message(FATAL_ERROR "ADC-700 MPI compiler version/SHA differs from receipt")
  endif()
  if(NOT IS_ABSOLUTE "${MPI_CXX_COMPILER}")
    message(FATAL_ERROR "ADC-700 FindMPI compiler path must be absolute")
  endif()
  file(REAL_PATH "${MPI_CXX_COMPILER}" _baseline_mpi_compiler)
  file(REAL_PATH "${_mpi_compiler}" _receipt_mpi_compiler)
  if(NOT _baseline_mpi_compiler STREQUAL "${_receipt_mpi_compiler}" OR
     NOT _mpi_standard STREQUAL "${MPI_CXX_VERSION}")
    message(FATAL_ERROR "ADC-700 baseline MPI compiler/version differs from candidate receipt")
  endif()
  _adc700_json_array(_receipt_mpi_includes "${_receipt_json}" mpi include_dirs)
  _adc700_json_array(_receipt_mpi_compile_options "${_receipt_json}" mpi compile_options)
  _adc700_json_array(_receipt_mpi_compile_definitions "${_receipt_json}" mpi compile_definitions)
  _adc700_json_array(_receipt_mpi_link_options "${_receipt_json}" mpi link_options)
  _adc700_json_array(_receipt_mpi_libraries "${_receipt_json}" mpi link_libraries)
  _adc700_parse_file_rows(_receipt_mpi_headers _receipt_mpi_header_hashes
    "${_receipt_json}" mpi headers)
  _adc700_parse_file_rows(_receipt_mpi_library_paths _receipt_mpi_library_hashes
    "${_receipt_json}" mpi libraries)
  if(NOT _receipt_mpi_libraries OR
     NOT "${_receipt_mpi_libraries}" STREQUAL "${_receipt_mpi_library_paths}")
    message(FATAL_ERROR "ADC-700 MPI link_libraries differ from authenticated library rows")
  endif()
  _adc700_target_property(_target_mpi_includes_raw MPI::MPI_CXX INTERFACE_INCLUDE_DIRECTORIES)
  _adc700_target_property(_target_mpi_compile_options MPI::MPI_CXX INTERFACE_COMPILE_OPTIONS)
  _adc700_target_property(_target_mpi_compile_definitions MPI::MPI_CXX INTERFACE_COMPILE_DEFINITIONS)
  _adc700_target_property(_target_mpi_link_options MPI::MPI_CXX INTERFACE_LINK_OPTIONS)
  _adc700_target_property(_target_mpi_libraries_raw MPI::MPI_CXX INTERFACE_LINK_LIBRARIES)
  _adc700_normalize_directories(_target_mpi_includes ${_target_mpi_includes_raw})
  set(_target_mpi_libraries "")
  foreach(_library IN LISTS _target_mpi_libraries_raw)
    cmake_path(NORMAL_PATH _library OUTPUT_VARIABLE _library_normalized)
    list(APPEND _target_mpi_libraries "${_library_normalized}")
  endforeach()
  if(NOT "${_receipt_mpi_includes}" STREQUAL "${_target_mpi_includes}" OR
     NOT "${_receipt_mpi_compile_options}" STREQUAL "${_target_mpi_compile_options}" OR
     NOT "${_receipt_mpi_compile_definitions}" STREQUAL "${_target_mpi_compile_definitions}" OR
     NOT "${_receipt_mpi_link_options}" STREQUAL "${_target_mpi_link_options}" OR
     NOT "${_receipt_mpi_libraries}" STREQUAL "${_target_mpi_libraries}")
    message(FATAL_ERROR "ADC-700 MPI target properties differ from candidate receipt")
  endif()
  if(MPI_CXX_INCLUDE_DIRS)
    _adc700_normalize_directories(_mpi_variable_includes ${MPI_CXX_INCLUDE_DIRS})
    if(NOT "${_mpi_variable_includes}" STREQUAL "${_target_mpi_includes}")
      message(FATAL_ERROR "ADC-700 MPI_CXX_INCLUDE_DIRS differs from MPI::MPI_CXX")
    endif()
  endif()
  if(MPI_CXX_COMPILE_OPTIONS AND
     NOT "${MPI_CXX_COMPILE_OPTIONS}" STREQUAL "${_target_mpi_compile_options}")
    message(FATAL_ERROR "ADC-700 MPI_CXX_COMPILE_OPTIONS differs from MPI::MPI_CXX")
  endif()
  if(MPI_CXX_COMPILE_DEFINITIONS AND
     NOT "${MPI_CXX_COMPILE_DEFINITIONS}" STREQUAL "${_target_mpi_compile_definitions}")
    message(FATAL_ERROR "ADC-700 MPI_CXX_COMPILE_DEFINITIONS differs from MPI::MPI_CXX")
  endif()
  set(_mpi_variable_link_options "")
  if(MPI_CXX_LINK_FLAGS)
    separate_arguments(_mpi_variable_link_options NATIVE_COMMAND "${MPI_CXX_LINK_FLAGS}")
    if(NOT "${_mpi_variable_link_options}" STREQUAL "${_target_mpi_link_options}")
      message(FATAL_ERROR "ADC-700 MPI_CXX_LINK_FLAGS differs from MPI::MPI_CXX")
    endif()
  endif()
  if(MPI_CXX_LIBRARIES)
    set(_mpi_variable_libraries "")
    foreach(_library IN LISTS MPI_CXX_LIBRARIES)
      cmake_path(NORMAL_PATH _library OUTPUT_VARIABLE _library_normalized)
      list(APPEND _mpi_variable_libraries "${_library_normalized}")
    endforeach()
    if(NOT "${_mpi_variable_libraries}" STREQUAL "${_target_mpi_libraries}")
      message(FATAL_ERROR "ADC-700 MPI_CXX_LIBRARIES differs from MPI::MPI_CXX")
    endif()
  endif()
  set(_expected_mpi_headers "")
  foreach(_include IN LISTS _receipt_mpi_includes)
    if(EXISTS "${_include}/mpi.h")
      list(APPEND _expected_mpi_headers "${_include}/mpi.h")
    endif()
  endforeach()
  if(NOT "${_receipt_mpi_headers}" STREQUAL "${_expected_mpi_headers}")
    message(FATAL_ERROR "ADC-700 MPI header rows do not match target include directories")
  endif()
  _adc700_mpi_abi(_observed_mpi_abi "${_receipt_mpi_compiler}" "${_mpi_standard}"
    _receipt_mpi_compile_options _receipt_mpi_compile_definitions _receipt_mpi_link_options
    _receipt_mpi_includes _receipt_mpi_headers _receipt_mpi_header_hashes
    _receipt_mpi_library_paths _receipt_mpi_library_hashes)
  if(NOT _observed_mpi_abi STREQUAL "${_mpi_abi}")
    message(FATAL_ERROR "ADC-700 MPI ABI digest does not authenticate the receipt fields")
  endif()

  # Kokkos is header-driven in generated MODULEs.  Authenticate both defining headers, the ABI
  # material, and every imported-target option/definition/library—not merely include presence.
  _adc700_json_string(_receipt_kokkos_abi "${_receipt_json}" kokkos abi_sha256)
  _adc700_require_hex("${_receipt_kokkos_abi}" 64 "candidate Kokkos ABI SHA-256")
  _adc700_json_array(_receipt_kokkos_includes "${_receipt_json}" kokkos include_dirs)
  _adc700_json_array(_receipt_kokkos_compile_options "${_receipt_json}" kokkos compile_options)
  _adc700_json_array(_receipt_kokkos_compile_definitions "${_receipt_json}" kokkos compile_definitions)
  _adc700_json_array(_receipt_kokkos_link_options "${_receipt_json}" kokkos link_options)
  _adc700_json_array(_receipt_kokkos_libraries "${_receipt_json}" kokkos link_libraries)
  _adc700_parse_file_rows(_receipt_kokkos_headers _receipt_kokkos_header_hashes
    "${_receipt_json}" kokkos headers)
  _adc700_target_property(_target_kokkos_includes_raw Kokkos::kokkos INTERFACE_INCLUDE_DIRECTORIES)
  _adc700_target_property(_target_kokkos_compile_options Kokkos::kokkos INTERFACE_COMPILE_OPTIONS)
  _adc700_target_property(_target_kokkos_compile_definitions Kokkos::kokkos INTERFACE_COMPILE_DEFINITIONS)
  _adc700_target_property(_target_kokkos_link_options Kokkos::kokkos INTERFACE_LINK_OPTIONS)
  _adc700_target_property(_target_kokkos_libraries Kokkos::kokkos INTERFACE_LINK_LIBRARIES)
  _adc700_normalize_kokkos_directories(_target_kokkos_includes ${_target_kokkos_includes_raw})
  if(NOT "${_receipt_kokkos_includes}" STREQUAL "${_target_kokkos_includes}" OR
     NOT "${_receipt_kokkos_compile_options}" STREQUAL "${_target_kokkos_compile_options}" OR
     NOT "${_receipt_kokkos_compile_definitions}" STREQUAL "${_target_kokkos_compile_definitions}" OR
     NOT "${_receipt_kokkos_link_options}" STREQUAL "${_target_kokkos_link_options}" OR
     NOT "${_receipt_kokkos_libraries}" STREQUAL "${_target_kokkos_libraries}")
    message(FATAL_ERROR "ADC-700 Kokkos target properties differ from candidate receipt")
  endif()
  set(_expected_kokkos_headers "")
  foreach(_include IN LISTS _receipt_kokkos_includes)
    foreach(_name IN ITEMS Kokkos_Core.hpp KokkosCore_config.h)
      if(EXISTS "${_include}/${_name}")
        list(APPEND _expected_kokkos_headers "${_include}/${_name}")
      endif()
    endforeach()
  endforeach()
  if(NOT "${_receipt_kokkos_headers}" STREQUAL "${_expected_kokkos_headers}")
    message(FATAL_ERROR "ADC-700 Kokkos header rows do not match target include directories")
  endif()
  list(LENGTH _receipt_kokkos_headers _kokkos_header_count)
  if(NOT _kokkos_header_count EQUAL 2)
    message(FATAL_ERROR "ADC-700 Kokkos receipt must authenticate exactly two defining headers")
  endif()
  _adc700_kokkos_abi(_observed_kokkos_abi _receipt_kokkos_includes
    _receipt_kokkos_headers _receipt_kokkos_header_hashes)
  if(NOT _observed_kokkos_abi STREQUAL "${_receipt_kokkos_abi}")
    message(FATAL_ERROR "ADC-700 Kokkos ABI digest does not authenticate header bytes")
  endif()
  _adc700_json_array(_receipt_loader_definitions "${_receipt_json}" native_loader compile_definitions)
  if(NOT "${_receipt_loader_definitions}" STREQUAL "POPS_RUNTIME_SHARED_EXCEPTION_ABI")
    message(FATAL_ERROR "ADC-700 native-loader definitions differ from the closed ABI contract")
  endif()

  # Apply the requested production flags to the *actual* baseline target and compare the complete
  # target properties.  This catches both missing and extra flags; a subset-presence check is not a
  # valid attestation of the generated MODULE command line.
  _adc700_json_string(_requested_cxx "${_receipt_json}" requested cxx)
  _adc700_json_string(_requested_include "${_receipt_json}" requested include)
  _adc700_json_string(_requested_std "${_receipt_json}" requested std)
  if(NOT _requested_std STREQUAL "c++20" OR NOT _requested_cxx STREQUAL "${_nvcc_path}" OR
     NOT IS_ABSOLUTE "${_requested_include}" OR NOT IS_DIRECTORY "${_requested_include}")
    message(FATAL_ERROR "ADC-700 requested compiler/include/C++ standard is not exact")
  endif()
  _adc700_json_array(_requested_compile_flags "${_receipt_json}" requested compile_flags)
  _adc700_json_array(_requested_link_flags "${_receipt_json}" requested link_flags)
  if(NOT _requested_compile_flags OR NOT _requested_link_flags)
    message(FATAL_ERROR "ADC-700 requested compile/link flag arrays must be non-empty")
  endif()
  if(NOT TARGET adc700_program_cutover)
    message(FATAL_ERROR "ADC-700 baseline target must exist before toolchain attestation")
  endif()
  target_compile_options(adc700_program_cutover PRIVATE ${_requested_compile_flags})
  target_link_options(adc700_program_cutover PRIVATE ${_requested_link_flags})
  _adc700_target_property(_effective_compile_flags adc700_program_cutover COMPILE_OPTIONS)
  _adc700_target_property(_effective_link_flags adc700_program_cutover LINK_OPTIONS)
  if(NOT "${_effective_compile_flags}" STREQUAL "${_requested_compile_flags}" OR
     NOT "${_effective_link_flags}" STREQUAL "${_requested_link_flags}")
    message(FATAL_ERROR "ADC-700 requested flags differ from flags effectively applied to baseline")
  endif()
  set(POPS_ADC700_TOOLCHAIN_ATTESTED TRUE PARENT_SCOPE)
endfunction()
