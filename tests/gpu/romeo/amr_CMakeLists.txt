cmake_minimum_required(VERSION 3.16)
project(amrgpu CXX)
find_package(Kokkos REQUIRED)
# Exact-ranked AMR reflux ledger. The staged source is the final ND consumer,
# not the retired 2-D implementation.
foreach(t test_nd_flux_ledger)
  add_executable(${t} ${t}.cpp)
  target_include_directories(${t} PRIVATE ${POPS_INCLUDE})
  target_compile_definitions(${t} PRIVATE POPS_HAS_KOKKOS)
  target_link_libraries(${t} PRIVATE Kokkos::kokkos)
endforeach()
