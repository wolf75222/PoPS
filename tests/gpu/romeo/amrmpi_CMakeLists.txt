cmake_minimum_required(VERSION 3.21)
project(amrmpi LANGUAGES CXX)

if(NOT POPS_ROOT)
  message(FATAL_ERROR
    "POPS_ROOT must name a complete staged PoPS source tree "
    "(CMakeLists.txt + cmake/ + include/ + src/)")
endif()
get_filename_component(POPS_ROOT "${POPS_ROOT}" ABSOLUTE)
foreach(_required
    CMakeLists.txt
    cmake/PopsDevTooling.cmake
    include/pops_headers.manifest
    src/CMakeLists.txt
    src/runtime/builders/seam_combinations.cmake)
  if(NOT EXISTS "${POPS_ROOT}/${_required}")
    message(FATAL_ERROR "POPS_ROOT is incomplete: missing ${POPS_ROOT}/${_required}")
  endif()
endforeach()

# Configure the real PoPS dependency contract but keep its Python module and test fleet out of this
# dedicated ROMEO build.  The root deliberately omits src/ when both consumers are disabled, so add
# the canonical runtime manifest explicitly instead of repeating its source/seam list here.
set(POPS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(POPS_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(POPS_USE_KOKKOS ON CACHE BOOL "" FORCE)
set(POPS_USE_MPI ON CACHE BOOL "" FORCE)
set(POPS_USE_HDF5 OFF CACHE BOOL "" FORCE)
add_subdirectory("${POPS_ROOT}" pops EXCLUDE_FROM_ALL)
set(POPS_CXX_STD 20)
add_subdirectory("${POPS_ROOT}/src" pops-runtime EXCLUDE_FROM_ALL)

add_executable(amrmpi_integrated amrmpi_integrated.cpp)
target_compile_features(amrmpi_integrated PRIVATE cxx_std_20)
# Linking the canonical OBJECT target embeds every AMR runtime unit and every generated seam while
# preserving the exact Kokkos/MPI/shared-exception usage requirements owned by PoPS.
target_link_libraries(amrmpi_integrated PRIVATE pops_runtime_amr)
