cmake_minimum_required(VERSION 3.21)
project(amrmpi LANGUAGES CXX)

if(NOT POPS_ROOT)
  message(FATAL_ERROR
    "POPS_ROOT must name a complete staged PoPS source tree "
    "(CMakeLists.txt + cmake/ + include/ + src/)")
endif()
get_filename_component(POPS_ROOT "${POPS_ROOT}" ABSOLUTE)
foreach(_required
    CMakeLists.txt
    cmake/PopsDevTooling.cmake
    include/pops_headers.manifest
    src/CMakeLists.txt)
  if(NOT EXISTS "${POPS_ROOT}/${_required}")
    message(FATAL_ERROR "POPS_ROOT is incomplete: missing ${POPS_ROOT}/${_required}")
  endif()
endforeach()

# Configure the real PoPS dependency contract but keep its Python module and test fleet out of this
# dedicated ROMEO build.  The root deliberately omits src/ when both consumers are disabled, so add
# the canonical runtime manifest explicitly instead of repeating its source/seam list here.
set(POPS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(POPS_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(POPS_USE_KOKKOS ON CACHE BOOL "" FORCE)
set(POPS_USE_MPI ON CACHE BOOL "" FORCE)
set(POPS_USE_HDF5 OFF CACHE BOOL "" FORCE)
add_subdirectory("${POPS_ROOT}" pops EXCLUDE_FROM_ALL)
set(POPS_CXX_STD 20)
add_subdirectory("${POPS_ROOT}/src" pops-runtime EXCLUDE_FROM_ALL)

function(add_amr_v5_program_fixture name block)
  add_library(${name} MODULE amrmpi_integrated.cpp)
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_definitions(${name} PRIVATE
    POPS_RUNTIME_SHARED_EXCEPTION_ABI
    POPS_TEST_AMR_V5_ARTIFACT=1
    POPS_TEST_AMR_V5_BLOCK0="${block}"
    POPS_TEST_AMR_V5_IDENTITY="${name}"
    POPS_TEST_AMR_V5_NUMERICAL=1)
  target_link_libraries(${name} PRIVATE pops::pops)
  if(APPLE)
    target_link_options(${name} PRIVATE -undefined dynamic_lookup)
  endif()
  set_target_properties(${name} PROPERTIES PREFIX "" POSITION_INDEPENDENT_CODE ON)
endfunction()

add_amr_v5_program_fixture(amrmpi_program_gas gas)
add_amr_v5_program_fixture(amrmpi_program_magnetic magnetic)

add_executable(amrmpi_integrated amrmpi_integrated.cpp)
target_compile_features(amrmpi_integrated PRIVATE cxx_std_20)
target_compile_definitions(amrmpi_integrated PRIVATE
  POPS_TEST_AMR_V5_GAS_PROGRAM="$<TARGET_FILE:amrmpi_program_gas>"
  POPS_TEST_AMR_V5_MAGNETIC_PROGRAM="$<TARGET_FILE:amrmpi_program_magnetic>")
set_target_properties(amrmpi_integrated PROPERTIES ENABLE_EXPORTS ON)
add_dependencies(amrmpi_integrated amrmpi_program_gas amrmpi_program_magnetic)
# Linking the canonical OBJECT target embeds the exact-ranked AMR runtime while preserving the
# Kokkos/MPI/shared-exception usage requirements owned by PoPS.
target_link_libraries(amrmpi_integrated PRIVATE pops_runtime_amr)
